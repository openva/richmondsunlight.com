<VirtualHost *:80>
  Protocols h2 http/1.1
  ServerName staging.richmondsunlight.com
  DocumentRoot /var/www/staging.richmondsunlight.com/htdocs
  RewriteMap tolower int:tolower
  ErrorLog ${APACHE_LOG_DIR}/staging.richmondsunlight.com.error.log
  CustomLog ${APACHE_LOG_DIR}/staging.richmondsunlight.com.access.log combined
</VirtualHost>
<Directory "/var/www/staging.richmondsunlight.com">
	AllowOverride All
	# Password-protect staging. This block covers both the HTTP and HTTPS (Certbot) vhosts.
	# Create the password file with: sudo htpasswd -c /etc/apache2/.htpasswd-staging <username>
	AuthType Basic
	AuthName "Richmond Sunlight staging"
	AuthUserFile /etc/apache2/.htpasswd-staging
	Require valid-user
</Directory>
